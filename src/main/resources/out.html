package com.example.stresstest;

import java.util.*;
import java.io.*;
import java.util.function.Function;

/**
 * Full feature stress test file.
 * Covers most Java syntax elements for lexer validation.
 */
@SuppressWarnings("unused")
public class MegaStressTest<T extends Number & Comparable<T>> {

    public static final int MAX_VALUE = 1_000_000;
    private static final String TEXT = "Hello \"world\"\nNext line test";

    private T value;
    private List<String> items = new ArrayList<>();

    public MegaStressTest(T value) {
        this.value = value;
    }

    @Deprecated
    public static void main(String[] args) throws Exception {

        MegaStressTest<Integer> test = new MegaStressTest<>(42);

        int a = 10;
        int b = 20;
        double d = 3.14159;
        long big = 1_000_000L;

        // arithmetic + logic
        int result = (a + b - 5) * 3 / 2 % 7;
        boolean logic = a > b && b <= 20 || a != 0;

        // bitwise
        int bits = a << 2 >> 1 ^ 5 & 3 | 8;

        // numbers edge cases
        float f = 1.23f;
        double sci = 1.2e10;
        int hex = 0xFF;
        int bin = 0b1010;

        // strings
        String s1 = "simple string";
        String s2 = "escape: \" \\ \n \t end";
        String s3 = """
                multiline
                text block
                """;

        // arrays
        int[] arr = {1, 2, 3, 4, 5};

        // for loop
        for (int i = 0; i < arr.length; i++) {
            System.out.println("index: " + i + " value: " + arr[i]);
        }

        // enhanced for
        for (int x : arr) {
            if (x % 2 == 0) continue;
            System.out.println("odd: " + x);
        }

        // while loop
        int i = 0;
        while (i < 3) {
            i++;
        }

        // do-while
        do {
            i--;
        } while (i > 0);

        // switch
        switch (a) {
            case 10 -> System.out.println("ten");
            case 20 -> System.out.println("twenty");
            default -> System.out.println("other");
        }

        // lambda
        Function<Integer, Integer> square = x -> x * x;
        System.out.println(square.apply(5));

        // collections
        List<String> list = new ArrayList<>();
        list.add("A");
        list.add("B");
        list.add("C");

        Map<String, Integer> map = new HashMap<>();
        map.put("one", 1);
        map.put("two", 2);

        for (String key : map.keySet()) {
            System.out.println(key + " -> " + map.get(key));
        }

        // nested scopes
        {
            int local = 123;
            System.out.println(local);
        }

        // annotations
        deprecatedMethod();

        // generics
        List<List<String>> nested = new ArrayList<>();
        nested.add(Arrays.asList("x", "y", "z"));

        // try-catch
        try {
            int x = 10 / 0;
        } catch (ArithmeticException e) {
            System.out.println("Error: " + e.getMessage());
        } finally {
            System.out.println("cleanup");
        }

        // operators chaos
        boolean test = (a == b) != false;
        int shift = (a << 1) + (b >> 2);

        System.out.println(TEXT + result + logic + test + shift);
    }

    @SafeVarargs
    public final void addItems(T... values) {
        for (T v : values) {
            items.add(String.valueOf(v));
        }
    }

    private static void deprecatedMethod() {
        // intentionally empty method
    }

    class InnerClass {
        void run() {
            System.out.println("inner " + value);
        }
    }
}
